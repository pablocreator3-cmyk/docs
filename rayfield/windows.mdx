local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

-- Creación de la Ventana Principal
local Window = Rayfield:CreateWindow({
   Name = "pablo_DTH Hub | Muscle Legends",
   LoadingTitle = "Cargando Potencia...",
   LoadingSubtitle = "por pablo_DTH",
   ConfigurationSaving = {
      Enabled = true,
      FolderName = "pablo_DTH_ML"
   },
   Theme = "DarkRed" -- Estética Rojo y Negro
})

-- Variables de estado
getgenv().autoWeight = false
getgenv().autoPunch = false

-- Pestaña de Entrenamiento
local FarmTab = Window:CreateTab("Auto Farm", 4483345998)

FarmTab:CreateSection("Entrenamiento de Fuerza")

-- Toggle para PESAS
FarmTab:CreateToggle({
   Name = "Auto-Equipar Pesa + AutoClick",
   CurrentValue = false,
   Flag = "TogglePesa",
   Callback = function(Value)
      getgenv().autoWeight = Value
      if Value then
         spawn(function()
            while getgenv().autoWeight do
               local char = game.Players.LocalPlayer.Character
               local bp = game.Players.LocalPlayer.Backpack
               -- Busca la pesa en inventario o mano
               local tool = bp:FindFirstChild("Weight") or char:FindFirstChild("Weight")
               
               if tool then
                  char.Humanoid:EquipTool(tool)
                  tool:Activate()
               end
               task.wait(0.01) -- Velocidad del click
            end
         end)
      end
   end,
})

-- Toggle para PUÑOS
FarmTab:CreateToggle({
   Name = "Auto-Equipar Puños + AutoClick",
   CurrentValue = false,
   Flag = "TogglePuno",
   Callback = function(Value)
      getgenv().autoPunch = Value
      if Value then
         spawn(function()
            while getgenv().autoPunch do
               local char = game.Players.LocalPlayer.Character
               local bp = game.Players.LocalPlayer.Backpack
               -- Busca el puño en inventario o mano
               local tool = bp:FindFirstChild("Punch") or char:FindFirstChild("Punch")
               
               if tool then
                  char.Humanoid:EquipTool(tool)
                  tool:Activate()
               end
               task.wait(0.01)
            end
         end)
      end
   end,
})

-- Pestaña de Personaje (Extras)
local PlayerTab = Window:CreateTab("Personaje", 4483345998)

PlayerTab:CreateSlider({
   Name = "Velocidad de Caminado",
   Range = {16, 500},
   Increment = 10,
   Suffix = "Speed",
   CurrentValue = 16,
   Flag = "SpeedSlider",
   Callback = function(Value)
      game.Players.LocalPlayer.Character.Humanoid.WalkSpeed = Value
   end,
})

PlayerTab:CreateButton({
   Name = "Resetear Personaje",
   Callback = function()
      game.Players.LocalPlayer.Character:BreakJoints()
   end,
})

-- Notificación de Inicio
Rayfield:Notify({
   Title = "pablo_DTH Hub Cargado",
   Content = "Script listo para Muscle Legends",
   Duration = 5,
   Image = 4483345998,
})
