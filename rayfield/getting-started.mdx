local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
   Name = "pablo_DTH Hub | Muscle Legends",
   LoadingTitle = "Cargando Interfaz Roja...",
   LoadingSubtitle = "por pablo_DTH",
   ConfigurationSaving = {
      Enabled = true,
      FolderName = "pablo_DTH_Config"
   },
   Theme = "DarkRed" -- Esto activa el tema oscuro con acentos rojos
})

-- Variables
getgenv().autoWeight = false
getgenv().autoPunch = false

-- Pestaña Principal
local MainTab = Window:CreateTab("Farm Principal", 4483345998)

MainTab:CreateSection("Entrenamiento Automático")

-- Toggle para Pesas
MainTab:CreateToggle({
   Name = "Auto Pesas + Auto Click",
   CurrentValue = false,
   Flag = "TogglePesas",
   Callback = function(Value)
      getgenv().autoWeight = Value
      if Value then
         spawn(function()
            while getgenv().autoWeight do
               local char = game.Players.LocalPlayer.Character
               local bp = game.Players.LocalPlayer.Backpack
               local weight = bp:FindFirstChild("Weight") or char:FindFirstChild("Weight")
               
               if weight then
                  char.Humanoid:EquipTool(weight)
                  weight:Activate()
               end
               task.wait(0.05)
            end
         end)
      end
   end,
})

-- Toggle para Puños
MainTab:CreateToggle({
   Name = "Auto Puños + Auto Click",
   CurrentValue = false,
   Flag = "TogglePunos",
   Callback = function(Value)
      getgenv().autoPunch = Value
      if Value then
         spawn(function()
            while getgenv().autoPunch do
               local char = game.Players.LocalPlayer.Character
               local bp = game.Players.LocalPlayer.Backpack
               local punch = bp:FindFirstChild("Punch") or char:FindFirstChild("Punch")
               
               if punch then
                  char.Humanoid:EquipTool(punch)
                  punch:Activate()
               end
               task.wait(0.05)
            end
         end)
      end
   end,
})

Rayfield:Notify({
   Title = "pablo_DTH Hub Cargado",
   Content = "Disfruta del script de Muscle Legends",
   Duration = 5,
   Image = 4483345998,
   Actions = {
      Ignore = {
         Name = "Entendido",
         Callback = function() end
      },
   },
})
